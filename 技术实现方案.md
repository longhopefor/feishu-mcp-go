#### Token 管理实现

```go
type TokenCache interface {
    Get(key string) (*TokenInfo, bool)
    Set(key string, token *TokenInfo, ttl time.Duration)
    Delete(key string)
}

type TokenInfo struct {
    AccessToken string
    ExpiresAt   time.Time
    TokenType   string
}

type tokenManager struct {
    client  *feishuClient
    cache   TokenCache
    mutex   sync.RWMutex
}

func (tm *tokenManager) GetValidToken(ctx context.Context) (string, error) {
    tm.mutex.RLock()
    if token, exists := tm.cache.Get("app_access_token"); exists {
        if time.Now().Before(token.ExpiresAt.Add(-5 * time.Minute)) {
            tm.mutex.RUnlock()
            return token.AccessToken, nil
        }
    }
    tm.mutex.RUnlock()
    
    tm.mutex.Lock()
    defer tm.mutex.Unlock()
    
    // 重新检查避免并发问题
    if token, exists := tm.cache.Get("app_access_token"); exists {
        if time.Now().Before(token.ExpiresAt.Add(-5 * time.Minute)) {
            return token.AccessToken, nil
        }
    }
    
    return tm.refreshToken(ctx)
}
```

### 2.4 批量操作优化实现

#### 批量创建块工具

```go
type BatchCreateBlocksTool struct {
    client FeishuClient
    logger Logger
}

func (t *BatchCreateBlocksTool) Execute(ctx context.Context, args map[string]interface{}) (*ToolResult, error) {
    // 解析参数
    blocks, err := t.parseBlocks(args["blocks"])
    if err != nil {
        return t.errorResult(fmt.Sprintf("参数解析失败: %v", err)), nil
    }
    
    // 分批处理 (每批最多50个块)
    const batchSize = 50
    var allResults []CreateBlockResult
    
    for i := 0; i < len(blocks); i += batchSize {
        end := i + batchSize
        if end > len(blocks) {
            end = len(blocks)
        }
        
        batch := blocks[i:end]
        results, err := t.createBlockBatch(ctx, args, batch)
        if err != nil {
            return t.errorResult(fmt.Sprintf("批次 %d 处理失败: %v", i/batchSize+1, err)), nil
        }
        
        allResults = append(allResults, results...)
    }
    
    return &ToolResult{
        Content: []ContentBlock{{
            Type: "text",
            Text: fmt.Sprintf("批量创建成功! 共创建 %d 个块", len(allResults)),
        }},
    }, nil
}

func (t *BatchCreateBlocksTool) createBlockBatch(ctx context.Context, args map[string]interface{}, blocks []BlockConfig) ([]CreateBlockResult, error) {
    req := &CreateBlocksRequest{
        DocumentID:    args["documentId"].(string),
        ParentBlockID: args["parentBlockId"].(string),
        Index:         int(args["index"].(float64)),
        Blocks:        blocks,
    }
    
    resp, err := t.client.CreateBlocks(ctx, req)
    if err != nil {
        return nil, err
    }
    
    return resp.Results, nil
}
```

```go
// 并发控制器
type ConcurrencyController struct {
    semaphore chan struct{}
    maxConcurrent int
}

func NewConcurrencyController(maxConcurrent int) *ConcurrencyController {
    return &ConcurrencyController{
        semaphore:     make(chan struct{}, maxConcurrent),
        maxConcurrent: maxConcurrent,
    }
}

func (cc *ConcurrencyController) Acquire() {
    cc.semaphore <- struct{}{}
}

func (cc *ConcurrencyController) Release() {
    <-cc.semaphore
}

// 在工具执行中使用
func (t *SomeTool) Execute(ctx context.Context, args map[string]interface{}) (*ToolResult, error) {
    t.concurrencyController.Acquire()
    defer t.concurrencyController.Release()
    
    // 执行具体逻辑
    return t.doExecute(ctx, args)
}
```

### 3.2 缓存机制

```go
// 多层缓存实现
type CacheLayer struct {
    l1Cache *bigcache.BigCache  // 内存缓存
    l2Cache *redis.Client       // Redis缓存 (可选)
    config  *CacheConfig
}

type CacheConfig struct {
    L1TTL           time.Duration
    L2TTL           time.Duration
    MaxMemoryMB     int
    EvictionPolicy  string
    CompressEnabled bool
}

func (cl *CacheLayer) Get(key string) ([]byte, error) {
    // 先查 L1 缓存
    if data, err := cl.l1Cache.Get(key); err == nil {
        return data, nil
    }
    
    // 再查 L2 缓存
    if cl.l2Cache != nil {
        if data, err := cl.l2Cache.Get(context.Background(), key).Bytes(); err == nil {
            // 回填 L1 缓存
            cl.l1Cache.Set(key, data)
            return data, nil
        }
    }
    
    return nil, ErrCacheNotFound
}

func (cl *CacheLayer) Set(key string, value []byte, ttl time.Duration) error {
    // 设置 L1 缓存
    if err := cl.l1Cache.Set(key, value); err != nil {
        return err
    }
    
    // 设置 L2 缓存
    if cl.l2Cache != nil {
        return cl.l2Cache.Set(context.Background(), key, value, ttl).Err()
    }
    
    return nil
}
```

### 4.1 统一错误处理

```go
// 错误类型定义
type FeishuError struct {
    Code    int    `json:"code"`
    Message string `json:"msg"`
    Data    interface{} `json:"data,omitempty"`
}

func (e *FeishuError) Error() string {
    return fmt.Sprintf("飞书API错误 [%d]: %s", e.Code, e.Message)
}

// 错误处理中间件
type ErrorHandler struct {
    logger Logger
}

func (eh *ErrorHandler) HandleError(err error) *ToolResult {
    switch e := err.(type) {
    case *FeishuError:
        return eh.handleFeishuError(e)
    case *ValidationError:
        return eh.handleValidationError(e)
    default:
        return eh.handleGenericError(e)
    }
}

func (eh *ErrorHandler) handleFeishuError(err *FeishuError) *ToolResult {
    eh.logger.Error("飞书API错误", "code", err.Code, "message", err.Message)
    
    var userMessage string
    switch err.Code {
    case 99991663:
        userMessage = "应用无权限访问该文档，请检查文档授权设置"
    case 99991664:
        userMessage = "应用不在该文档的可用范围内"
    case 230005:
        userMessage = "文档不存在或已被删除"
    default:
        userMessage = fmt.Sprintf("操作失败: %s", err.Message)
    }
    
    return &ToolResult{
        IsError: true,
        Content: []ContentBlock{{
            Type: "text",
            Text: userMessage,
        }},
    }
}
```

```go
// 指标收集器
type MetricsCollector struct {
    requestCount     prometheus.CounterVec
    requestDuration  prometheus.HistogramVec
    errorCount       prometheus.CounterVec
    cacheHitRate     prometheus.GaugeVec
}

func NewMetricsCollector() *MetricsCollector {
    mc := &MetricsCollector{
        requestCount: *prometheus.NewCounterVec(
            prometheus.CounterOpts{
                Name: "feishu_mcp_requests_total",
                Help: "Total number of requests",
            },
            []string{"tool", "status"},
        ),
        requestDuration: *prometheus.NewHistogramVec(
            prometheus.HistogramOpts{
                Name:    "feishu_mcp_request_duration_seconds",
                Help:    "Request duration in seconds",
                Buckets: prometheus.DefBuckets,
            },
            []string{"tool"},
        ),
    }
    
    prometheus.MustRegister(mc.requestCount, mc.requestDuration, mc.errorCount, mc.cacheHitRate)
    return mc
}

// 在工具执行中记录指标
func (t *BaseTool) ExecuteWithMetrics(ctx context.Context, args map[string]interface{}) (*ToolResult, error) {
    start := time.Now()
    defer func() {
        duration := time.Since(start).Seconds()
        t.metrics.requestDuration.WithLabelValues(t.Name()).Observe(duration)
    }()
    
    result, err := t.Execute(ctx, args)
    
    status := "success"
    if err != nil || (result != nil && result.IsError) {
        status = "error"
        t.metrics.errorCount.WithLabelValues(t.Name()).Inc()
    }
    
    t.metrics.requestCount.WithLabelValues(t.Name(), status).Inc()
    
    return result, err
}
```
